options:
  -h, --help            show this help message and exit
  --cards CARDS
  --srt SRT
  --voiceover VOICEOVER
  --out OUT
  --audio AUDIO
  --strict
  --dry-run
  --grain GRAIN         film grain + vignette intensity 0-20 (O-2210)
  --bg BG               override video bg color (e.g. 0x0a0a0d)
  --bgvideo BGVIDEO     real-footage background video (O-20260924-1115);
                        looped, scaled to frame, replaces the color source
  --no-cards            pure-documentary cut: AIGC notice + subs only
  --poster POSTER       after a successful render, export the cover frame to
                        this PNG (first card at full opacity; ignored with
                        --dry-run) - audit O-1043 R1
  --series-badge        visual-spec v1.2 S5.5 T2: top-right BigStream series
                        badge, constant for the whole video
  --series-id SERIES_ID
                        S5.5 T5 series/episode id (e.g. 'BS-001 EP.01'); rides
                        the badge line, required by --series-intro
  --series-intro        S5.5 T1: 1.5-2s black series intro frame (episode id +
                        brand line, accent color)
  --intro-s INTRO_S     T1 intro length in seconds, clamped to 1.5-2.0
  --h1-glow             S4.5(1): restrained 2px accent glow edge on H1
  --scanlines           S4.5(3): scanline ambience layer (5 percent, 8 percent
                        spec cap)
  --sys-status          S4.5(3): per-cue machine telemetry line sys.beat=NN
                        t=MM:SS
