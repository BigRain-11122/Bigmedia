 function fetchJ(path,opts,ms){
  return new Promise(function(res,rej){
   var ctl=('AbortController' in window)?(new AbortController()):null;
   var t=setTimeout(function(){if(ctl)ctl.abort();rej(new Error('timeout'));},ms||4500);
   var init=opts||{};
   if(ctl)init.signal=ctl.signal;
   fetch(API+path,init).then(function(r){
    clearTimeout(t);
    if(!r.ok){rej(new Error('http'+r.status));return;}
    r.json().then(function(j){res(j);},function(e){rej(e);});
   },function(e){clearTimeout(t);rej(e);});
  });
 }

  rows.forEach(function(r){
   var st=(r.status==='done')?('<span class="rc-badge rc-b-done">✅ 已入册 '+(r.u||'')+'</span>')
        :((r.mode==='now')?'<span class="rc-badge rc-b-nb">⚡ 待执行</span>':'<span class="rc-badge rc-b-td">📝 排队</span>');
   var t=String(r.tx||'');if(t.length>76)t=t.slice(0,76)+'…';

  });

 }

  fetchJ('/api/cmd',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(payload)},6000)
   .then(function(j){
    if(!j||!j.ok)throw new Error((j&&j.err)||'bad');

 }
